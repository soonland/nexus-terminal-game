- Style: arrow functions only, semicolons, single quotes, trailing commas, 100-col; `eslint --fix` runs on commit and strips casts it judges unnecessary, so run `pnpm tsc -b` and `pnpm lint` before committing. Prettier reformats `anchorNodes.ts`: run `pnpm format` after inserting content.
- Pre-commit checks, in order, before every commit: `pnpm format`, `pnpm tsc -b`, `pnpm build`, `pnpm lint`, `pnpm test:coverage`, `pnpm knip`. Per-file coverage must stay ≥ 75%.

## Helper used by Tasks 2–4: `insert_lore.py`

Save this in the scratchpad (not in the repo) as `insert_lore.py`; each task's script loads it with `exec(open('<scratchpad>/insert_lore.py').read())` (replace `<scratchpad>` with the real scratchpad directory) and applies its own documents. It reads the approved text from the outline and edits `src/data/anchorNodes.ts` with asserted, anchored replacements.

```python
import json
import re

OUTLINE = 'docs/superpowers/specs/2026-10-03-lore-pass-design.md'
DATA = 'src/data/anchorNodes.ts'

outline = open(OUTLINE).read()
data = open(DATA).read()

def dumps(value):
    return json.dumps(value, ensure_ascii=False)


FENCE = chr(96) * 3  # three backticks, built here so this file can live inside a markdown fence


def block(doc_id, index=0):
    """Return the index-th fenced block of the '### <doc_id>.' section of the outline."""
    section = re.search(rf'### {doc_id}\..*?(?=\n### |\n---|\n## )', outline, re.S)
    assert section, f'missing section {doc_id}'
    blocks = re.findall(FENCE + r'\n(.*?)' + FENCE, section.group(0), re.S)
    return blocks[index].rstrip('\n')


def ts_file(name, path, file_type, content, access, exfiltrable=True):
    return (
        '      {\n'
        f'        name: {dumps(name)},\n'
        f'        path: {dumps(path)},\n'
        f'        type: {dumps(file_type)},\n'
        f'        content: {dumps(content)},\n'
        f'        exfiltrable: {"true" if exfiltrable else "false"},\n'
        f'        accessRequired: {dumps(access)},\n'
        '      },\n'
    )


def node_span(node_id):
    start = data.index(f"id: '{node_id}'")
    nxt = data.find("\n  {\n    id: '", start + 1)
    return start, (nxt if nxt != -1 else len(data))


def append_file(node_id, file_ts):
    global data
    start, end = node_span(node_id)
    files = data.index('files: [', start)
    assert files < end, node_id
    depth, i = 0, files + len('files: ')
    while True:  # bracket-match the files array
        if data[i] == '[':
            depth += 1
        elif data[i] == ']':
            depth -= 1
            if depth == 0:
                break
        i += 1
    data = data[:i].rstrip() + '\n' + file_ts + '    ' + data[i:]


def replace_null_content(node_id, file_name, new_content):
    """Give an AI-generated (content: null) file authored content."""
    global data
    start, end = node_span(node_id)
    name_at = data.index(f"name: '{file_name}'", start)
    assert name_at < end, file_name
    null_at = data.index('content: null', name_at)
    assert null_at - name_at < 400, file_name
    data = data[:null_at] + f'content: {dumps(new_content)}' + data[null_at + len('content: null'):]


def replace_content_literal(node_id, file_name, new_content):
    """Replace an existing authored content string."""
    global data
    start, end = node_span(node_id)
    name_at = data.index(f"name: '{file_name}'", start)
    pattern = re.compile(r'content:\s*(\'(?:[^\'\\]|\\.)*\'|"(?:[^"\\]|\\.)*"),', re.S)
    match = pattern.search(data, name_at)
    assert match and match.start() - name_at < 400, file_name
    data = data[:match.start()] + f'content: {dumps(new_content)},' + data[match.end():]


def replace_once(old, new):
    global data
    assert data.count(old) == 1, (old, data.count(old))
    data = data.replace(old, new)


def save():
    open(DATA, 'w').write(data)
```
